declare upper;

#Hint: Uses a linear regression to predict a fair value for the symbol on the chart. <br>Beware of trends : High R but no true correlation. <br>Beware of dividends : An ex-date on this or the other symbol will change the predicted values. If the dividend is super high and / or the ex-date between the two is not close to the other, the predicted price will vary.

# Regression Settings ...
input symbol = "SPX"; #Hint symbol: Other stock, ETF, futures contract, index, or whatever to use to predict the fair value.
input window = 252; #Hint window: Lookback window for data. Recommended to be above 252 (1 Trading Year). This is what the model will be traded on. The window does NOT include today. So a window of 252 will use a bar from 253 trading days ago.
input extra_out_of_sample_push = 0; #Hint extra_out_of_sample_push: The model is, by default, trained on the last "window" days to then predict today. But, if you would like you can add more days to not include. So: by default it's [-window, yesterday] but you can do [-window - extra, yesterday - extra]. For example: A window of 252 would be: [-253, -1], but if you add 20 days it would be: [-273, -21].
def oosp = max(extra_out_of_sample_push+1,1);

# Predicted Value Line Settings ...
input show_line = yes; #Hint show_line: Show history of predicted values.
input show_extrapolation = no; #Hint show_extrapolation: Change the color of the predicted value line to show when the model is extrapolating versus interpolating.

# Label Settings ...
input show_predicted_price_label = yes; #Hint show_predicted_price_label: Creates a Label to display the Predicted Price.
input show_r_label = yes; #Hint show_r_label: Creates a Label to display Correlation.
input show_premium_discount_label = yes; #Hint show_premium_discount_label: Creates a Label to display the premium or discount compared to the predicted fair value.
input show_beta_label = yes; #Hint show_beta_label: Creates a Label to display the Beta.

# Make sure the period is at least a DAY.
def p = if GetAggregationPeriod() < AggregationPeriod.DAY then AggregationPeriod.DAY else GetAggregationPeriod();

def x = Log(close(symbol = symbol, period = p));
def y = Log(close(period = p));

def xmean = Average(x, window);
def ymean = Average(y, window);

def xdist = x - xmean;
def ydist = y - ymean;

def cov0 = (
    fold i0 = oosp to window + oosp with n0 = 0 do
        n0 + ((x[i0] - xmean[oosp]) * (y[i0] - ymean[oosp]))
);

def cov = (cov0) / (window - 1);

def xdistsumsqr = (
    fold i1 = oosp to window + oosp with n1 = 0 do
        n1 + ((x[i1] - xmean[oosp]) * (x[i1] - xmean[oosp]))
);
def ydistsumsqr = (
    fold i2 = oosp to window + oosp with n2 = 0 do
        n2 + ((y[i2] - ymean[oosp]) * (y[i2] - ymean[oosp]))
);
def xvar = xdistsumsqr / (window - 1);
def yvar = ydistsumsqr / (window - 1);

def xstd = Sqrt(xvar);
def ystd = Sqrt(yvar);

def corcoeff = cov / (xstd * ystd);
def beta = ystd / xstd;
def m = corcoeff * beta;
def b = ymean - (xmean * m);

def pred = Exp(m * x + b);
plot pred_plot = if show_line then pred else Double.NaN;
pred_plot.HideBubble();

# Extrapolation or Interpolation ?
def xmax = Highest(x, window);
def xmin = Lowest(x, window);
def interpolation = xmin[1] <= x and x <= xmax[1];
pred_plot.AssignValueColor(if interpolation or !show_extrapolation then CreateColor(050, 150, 250) else CreateColor(250, 150, 050));

# Labels ...
AddLabel(show_predicted_price_label, "Predicted Price: $" + Round(pred, 2), Color.BLACK, Location.TOP_LEFT);
AddLabel(show_r_label, "R: " + Round((AbsValue(corcoeff) * 100), 2) + "%  " + "R^2: " + Round((AbsValue(corcoeff*corcoeff) * 100), 2) + "%", Color.BLACK, Location.TOP_LEFT);
AddLabel(show_premium_discount_label, 
    (if close > pred then "Premium: " else "Discount: ") + 
        Round(100 * ((close / pred) - 1), 2) + "%",
Color.BLACK, Location.TOP_LEFT
);
AddLabel(show_beta_label, "Beta: " + Round(m, 3), Color.BLACK, Location.TOP_LEFT);

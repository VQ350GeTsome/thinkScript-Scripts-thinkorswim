# Clamp to [2, 32]
def o = if octaves > 32 then 32 else if octaves < 2 then 2 else octaves;
# Aprox quadratic weighting
def PI = 3.14159265359;
def w = ((PI*PI) / 6) - (1 / (o + 0.55));

# Get Harmonic Number
def h = 

         if quadratic_mode then w

    else if o ==  2 then 1.5
    else if o ==  3 then 1.83333333333

# Weighted sSums
def avg_f = 
    a1 + a2 / 2 + a3 / 3 + a4 / 4 + a5 / 5 + a6 / 6 + a7 / 7 + a8 / 8 + 
    a9 / 9 + aA / 10 + aB / 11 + aC / 12 + aD / 13 + aE / 14 + aF / 15 + 
    aG / 16 + aH / 17 + aI / 18 + aJ / 19 + aK / 20 + aL / 21 + aM / 22 +

    aU / 30 + aV / 31 + aW / 32
;

    a1 / o + a2 / (o - 1) + 
    a3 * (if o >  2 then 1 / (o -  2) else 0) +

    a5 * (if o >  4 then 1 / (o -  4) else 0) +

    aC * (if o > 11 then 1 / (o - 11) else 0) +
    aD * (if o > 12 then 1 / (o - 12) else 0) +
    aE * (if o > 13 then 1 / (o - 13) else 0) +
    aF * (if o > 14 then 1 / (o - 14) else 0) +
    aG * (if o > 15 then 1 / (o - 15) else 0) +
    aH * (if o > 16 then 1 / (o - 16) else 0) +
    aI * (if o > 17 then 1 / (o - 17) else 0) +
    aJ * (if o > 18 then 1 / (o - 18) else 0) +
    aK * (if o > 19 then 1 / (o - 19) else 0) +
    aL * (if o > 20 then 1 / (o - 20) else 0) +
    aM * (if o > 21 then 1 / (o - 21) else 0) +
    aN * (if o > 22 then 1 / (o - 22) else 0) +
    aO * (if o > 23 then 1 / (o - 23) else 0) +
    aP * (if o > 24 then 1 / (o - 24) else 0) +
    aQ * (if o > 25 then 1 / (o - 25) else 0) +
    aR * (if o > 26 then 1 / (o - 26) else 0) +
    aS * (if o > 27 then 1 / (o - 27) else 0) +
    aT * (if o > 28 then 1 / (o - 28) else 0) +
    aU * (if o > 29 then 1 / (o - 29) else 0) +
    aV * (if o > 30 then 1 / (o - 30) else 0) +
    aW * (if o > 31 then 1 / (o - 31) else 0)
;
def avg_q = 
    a1 + a2 / 4 + a3 / 9 + a4 / 16 + a5 / 25 + a6 / 32 + 
    a7 / 49 + a8 / 64 + a9 / 81 + aA / 100 + aB / 121 + 
    aC / 144 + aD / 169 + aE / 196 + aF / 225 + aG / 256 +

    aM / 484 + aN / 529 + aO / 576 + aP / 625 + aQ / 676 +

    a1 / 2 + a3 / 4 + a4 / 8 + a5 / 16 + a6 / 32 + a7 / 64 + 
    a8 / 128 + a9 / 256 + aA / 512 + aB / 1024 + aC / 2048 +
    aD / 4096 + aE / 8192 + aF / 16384 + aG / 32768 + aH / 65536 +
    aI / 131072

# Hints
#hint length: Raw Length. Real length is this divided by the <b>nth</b> harmonic number where <b>n = octaves</b>.
#hint octaves: How many moving averages to use. If you just want one octave, use a regular moving average of whatever type you picked. 32 is the max implemented, anything more than that is kinda useless. I personally only go up to 16.
#hint average_type: Simple & Weighted seems to produce the best results.
#hint quadratic_mode: Instead of linear weighting, use quadratic weighting ... this only makes the FMA more similar to a regular moving average ... but it's interesting nonetheless.
#hint display_regular: Plot the actual main moving average. Turn off if you only want the reverse.
#hint display_reverse: Plot the reverse weighted moving average. Great for crossovers.
#hint cloud_mode: Displays a cloud between the FMA & Reverse FMA ( RFMA ).
#hint show_ribbon: Turns off the regular FMA & RFMA, and instead displays the ribbon used to calculate them.
